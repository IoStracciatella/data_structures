As expressões *p++ e (*p)++ são diferentes porque, na primeira, incrementamos o ponteiro, fazendo-o apontar para a próxima posição de memória, e, na segunda, incrementamos o valor no bloco de memória. No caso do código que está no exercício, a saída é abc a = abc a = dbc, afinal, no segundo caso mudamos o valor da variável no bloco de memória

#include <stdio.h>

int main() {
    int i, soma = 0, elementos = 0, numeros[15];
    float media;
    
    for (i = 0; i < 15; i++) {
        scanf("%d", &numeros[i]);
    }
    
    for (i = 0; i < 15; i++) {
        if (i % 2 == 0) {
            printf("%d", numeros[i]);
            soma += numeros[i];
            elementos++;
        }
    }
    
    media = (float) soma / (float) elementos;
    
    printf("Média: %f", media);

    return 0;
}
