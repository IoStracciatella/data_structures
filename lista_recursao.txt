#include <stdio.h>

void conta_tempo(int n) {
    if (n > 0) {
        printf("%d ", n);

        conta_tempo(n - 1);

        printf("%d ", n);
    }
}

int main() {
    int n;

    scanf("%d", &n);

    conta_tempo(n);

    return 0;
}

int soma_digitos(int n)
{
    if (n < 10)
        return n;

    return (n % 10) + soma_digitos(n / 10);
}

void imprime_binario(int n)
{
    if (n < 2)
    {
        printf("%d", n);
        return;
    }

    imprime_binario(n / 2);
    printf("%d", n % 2);
}

int menor_elemento(int v[], int n)
{
    int menor;

    if (n == 1)
        return v[0];

    menor = menor_elemento(v + 1, n - 1);

    if (v[0] < menor)
        return v[0];

    return menor;
}
