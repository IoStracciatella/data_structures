#include <stdio.h>

void conta_tempo(int n) {
    if (n > 0) {
        printf("%d ", n);

        conta_tempo(n - 1);

        printf("%d ", n);
    }
}

int main() {
    int n;

    scanf("%d", &n);

    conta_tempo(n);

    return 0;
}

int soma_digitos(int n)
{
    if (n < 10)
        return n;

    return (n % 10) + soma_digitos(n / 10);
}
